\begin{frame}{Why this project?}
\begin{itemize}
  \item We do not work in finance: curious on \textbf{very noisy} data and \textbf{non-trivial spectral analysis} with extreme events.
  \item We reproduced a hedge-fund preprint: \textbf{Bouchaud et al., ``Riding Wavelets''} (jump classes via wavelets + Kernel PCA).
  \item Key motivation: \textbf{non-stationary markets} (volatility clustering, intraday patterns) so global templates (e.g., dictionary learning) fail.
  \item Goal: reproduce the full pipeline from the paper with \textbf{open data}, end-to-end, and test across \textbf{time scales} and \textbf{markets}.

\end{itemize}

\begin{frame}{Jump normalization (seasonality + local volatility)}
\scriptsize
\begin{itemize}
  \item Markets highly non-stationary = standardize returns to jump score $x(t)$
\end{itemize}

\vspace{0.15em}
\[
x(t)=\frac{r(t)}{f(t)\sigma(t)}.
\]

\vspace{0.25em}
\begin{fboxenv}{Seasonality $f_t$}
\small
\[
\begin{aligned}
r_t &= \log p_t-\log p_{t-\Delta}, \qquad \tau(t)=\text{time-of-day of }t,\\
m(\tau) &= \mathrm{median}\big\{|r_t|:\tau(t)=\tau\big\},\\
q &= \sqrt{\frac{1}{|\mathcal{T}|}\sum_{\tau\in\mathcal{T}} m(\tau)^2},\\
f_t &= \frac{m(\tau(t))}{q}, \qquad r^{\mathrm{des}}_t = \frac{r_t}{f_t}, \qquad (\text{daily: } f_t\equiv 1).
\end{aligned}
\]
\end{fboxenv}

\vspace{0.25em}
\begin{sigmaboxenv}{Local volatility $\sigma_t$}
\small
\[
\begin{aligned}
\mu_t &= \alpha r^{\mathrm{des}}_t+(1-\alpha)\mu_{t-1},\\
v_t &= \alpha\big(r^{\mathrm{des}}_t-\mu_t\big)^2+(1-\alpha)v_{t-1},\\
\sigma_t &= \sqrt{v_t},\qquad \alpha=\frac{2}{s+1}.
\end{aligned}
\]
\end{sigmaboxenv}

\end{frame}

  \begin{frame}{Wavelet features and interpretable directions}
    \small
    \begin{itemize}
      \item We align jump signs ($t=0$).
      \item We compute complex wavelet scattering.
    \end{itemize}
    \[
    \Phi(x)=\big(W_j\bar{x}(0),\; W_{j_2}\,|W_{j_1}x|(0)\big)_{1\le j\le J,\; 1\le j_1<j_2\le J}.
    \]
    \begin{itemize}
      \item We run Kernel PCA on feature vector $\Phi(x)$: $K(\Phi(x_i),\Phi(x_j))$.
    \end{itemize}
    \end{frame}
